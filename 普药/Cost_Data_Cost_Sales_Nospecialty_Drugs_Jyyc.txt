CREATE OR REPLACE PROCEDURE "public"."Cost_Data_Cost_Sales_Nospecialty_Drugs_Jyyc"(
    IN "p_target_date" varchar,
    INOUT "result" int4,
    INOUT "msg" text
)
AS $BODY$
DECLARE
    v_yesterday VARCHAR(10);
    v_cperiod VARCHAR(7);
    v_cperiod_year VARCHAR(4);
    v_start_date VARCHAR(10);
    v_last_month_period VARCHAR(7);
    v_source_count INT;
    v_sub_result INT4 DEFAULT 1;
    v_sub_msg TEXT DEFAULT '';
BEGIN
    result := 1;
    msg := 'Procedure initialized';

    IF COALESCE(BTRIM(p_target_date), '') = '' THEN
        result := 0;
        msg := 'Parameter p_target_date is required';
        RETURN;
    END IF;

    v_yesterday := TO_CHAR(CAST(p_target_date AS DATE) - INTERVAL '1 day', 'YYYY-MM-DD');
    v_cperiod := SUBSTRING(v_yesterday, 1, 7);
    v_cperiod_year := SPLIT_PART(v_cperiod, '-', 1);
    v_start_date := CONCAT(v_cperiod, '-01');
    v_last_month_period := TO_CHAR(CAST(v_start_date AS DATE) - INTERVAL '1 month', 'YYYY-MM');

    DELETE FROM public.mis_auto_jyycmx
    WHERE hesuanqijian = v_cperiod
      AND COALESCE(pinzhongleixing, '') = '普药非新特药';

    DROP TABLE IF EXISTS temp_ns_sale_ids;
    CREATE TEMP TABLE temp_ns_sale_ids AS
    SELECT
        p.id
    FROM mis_auto_ptpcr p
    LEFT JOIN public.mis_auto_jyycmx j
        ON j.hesuanqijian = v_cperiod
       AND j.srcid = CAST(p.id AS VARCHAR)
       AND COALESCE(j.pinzhongleixing, '') IN ('新特药', '骨干')
    LEFT JOIN public.mis_auto_xinteyao_lirun_mingxi xm
        ON xm.jisuanqijian = v_cperiod
       AND CAST(xm.srcid AS VARCHAR) = CAST(p.id AS VARCHAR)
    LEFT JOIN public.mis_auto_hlutv hl
        ON hl.cperiod = v_cperiod
       AND CAST(hl.saleID AS VARCHAR) = CAST(p.id AS VARCHAR)
    WHERE p.cperiod = v_cperiod
      AND CAST(p.dbilldate AS DATE) >= CAST(v_start_date AS DATE)
      AND CAST(p.dbilldate AS DATE) <= CAST(v_yesterday AS DATE)
      AND j.srcid IS NULL
      AND xm.srcid IS NULL
      AND hl.saleID IS NULL
    GROUP BY p.id;

    DROP TABLE IF EXISTS temp_xt_sale_ids;
    CREATE TEMP TABLE temp_xt_sale_ids AS
    SELECT DISTINCT
        CAST(j.srcid AS BIGINT) AS id
    FROM public.mis_auto_jyycmx j
    WHERE j.hesuanqijian = v_cperiod
      AND COALESCE(j.pinzhongleixing, '') = '新特药'
      AND j.srcid ~ '^[0-9]+$'
      AND CAST(j.chukudanjuriqi AS DATE) >= CAST(v_start_date AS DATE)
      AND CAST(j.chukudanjuriqi AS DATE) <= CAST(v_yesterday AS DATE);

    DROP TABLE IF EXISTS temp_shared_sale_ids;
    CREATE TEMP TABLE temp_shared_sale_ids AS
    SELECT id FROM temp_ns_sale_ids
    UNION
    SELECT id FROM temp_xt_sale_ids;

    SELECT COUNT(*)
    INTO v_source_count
    FROM temp_shared_sale_ids;

    IF COALESCE(v_source_count, 0) = 0 THEN
        result := 1;
        msg := CONCAT('No shared forecast source data found. Period: ', v_cperiod, ', range: ', v_start_date, ' to ', v_yesterday);
        RETURN;
    END IF;

    -- 直接使用 public.mis_auto_jyycmx 作为经营预测结果及工作表，不再创建同名临时表。

    DROP TABLE IF EXISTS temp_ldttj_customer_mode;
    CREATE TEMP TABLE temp_ldttj_customer_mode AS
    SELECT
        CASE
            WHEN EXISTS (
                SELECT 1
                FROM mis_auto_ldttj mas
                INNER JOIN mis_auto_ldttj_sub_dijiamingxi up
                    ON up.masid = mas.id
                WHERE mas.cperiod = v_last_month_period
                  AND mas.status = 1
                  AND mas.isvoid = 0
                  AND up.isvoid = 0
                  AND NULLIF(BTRIM(COALESCE(up.customer, '')), '') IS NOT NULL
            ) THEN 1
            ELSE 0
        END AS has_customer_dim;

    DROP TABLE IF EXISTS temp_mis_auto_ldttj;
    CREATE TEMP TABLE temp_mis_auto_ldttj AS
    SELECT *
    FROM (
        SELECT
            p.id AS invoice_id,
            COALESCE(CAST(NULLIF(TRIM(up.price::TEXT), '') AS NUMERIC(24,8)), 0) AS price,
            ROW_NUMBER() OVER (
                PARTITION BY p.id
                ORDER BY
                    CASE
                        WHEN mode.has_customer_dim = 1
                         AND NULLIF(BTRIM(COALESCE(up.customer, '')), '') = NULLIF(BTRIM(COALESCE(lbvrh.name, '')), '') THEN 1
                        ELSE 2
                    END,
                    up.id
            ) AS rn
        FROM mis_auto_ptpcr p
        INNER JOIN temp_ns_sale_ids s
            ON s.id = p.id
        CROSS JOIN temp_ldttj_customer_mode mode
        LEFT JOIN mis_auto_yuorb dep
            ON dep.pk_vid = p.cdeptvid
        LEFT JOIN mis_auto_lbvrh lbvrh
            ON lbvrh.pk_cust_sup_v = p.cinvoicecustid
        LEFT JOIN mis_auto_ldttj mas
            ON mas.cperiod = v_last_month_period
           AND mas.status = 1
           AND mas.isvoid = 0
        LEFT JOIN mis_auto_ldttj_sub_dijiamingxi up
            ON up.masid = mas.id
           AND up.cinvcode = p.material_code
           AND up.saledeptcode = dep.code
           AND (
                (
                    mode.has_customer_dim = 1
                    AND NULLIF(BTRIM(COALESCE(up.customer, '')), '') = NULLIF(BTRIM(COALESCE(lbvrh.name, '')), '')
                )
                OR (
                    mode.has_customer_dim = 0
                    AND NULLIF(BTRIM(COALESCE(up.customer, '')), '') IS NULL
                )
           )
           AND up.isvoid = 0
    ) matched_price
    WHERE rn = 1;

    DROP TABLE IF EXISTS temp_mis_auto_tbdua;
    CREATE TEMP TABLE temp_mis_auto_tbdua AS
    SELECT *
    FROM (
        SELECT
            p.id AS invoice_id,
            tb.measrate,
            ROW_NUMBER() OVER (
                PARTITION BY p.id
                ORDER BY
                    CASE WHEN RIGHT(COALESCE(p.material_spec, ''), 1) = tu.name THEN 1 ELSE 2 END,
                    tb.pk_measdoc
            ) AS rn
        FROM mis_auto_ptpcr p
        INNER JOIN temp_ns_sale_ids s
            ON s.id = p.id
        INNER JOIN mis_auto_rbyld r
            ON r.pk_material = p.cmaterialvid
        INNER JOIN mis_auto_tbdua tb
            ON r.pk_material = tb.pk_material
        LEFT JOIN mis_auto_tutyp tu
            ON tu.pk_measdoc = tb.pk_measdoc
    ) matched_unit
    WHERE rn = 1;

    INSERT INTO public.mis_auto_jyycmx (
        id, srcid, hesuanqijian, fapiaoqijian, chukudanjuriqi, jinshuipiaohao,
        csaleinvoiceid1, jinshuikaipiaoriqi, shijichukuriqi, kaipiaokehu,
        xiaoshoubumen, xiaoshoubumenduiyingdiqu, xiaoshouyewuyuan,
        xiaoshouqudao, qudaoyewuyuan, diqufenlei, shengshiqu, yetai,
        pinzhongleixing, pinzhong, pingzhenghao, wuliaobianma, wuliaomingcheng,
        guige, zhushuliang, jianshu, zhuhanshuidanjia, zhuhanshuijingjia,
        wushuijineheji, jiashuiheji, feiyongchongdijine, chongdiqianjine,
        zhekoue, cdq_je_bhs, danweidijia, dijiajine,
        cinvoicecustid, firstvbillcode, cmaterialvid, create_time, istatus
    )
    SELECT
        CAST(p.id AS BIGINT) AS id,
        CAST(p.id AS VARCHAR) AS srcid,
        v_cperiod AS hesuanqijian,
        p.cperiod AS fapiaoqijian,
        p.dbilldate AS chukudanjuriqi,
        p.vdef__1 AS jinshuipiaohao,
        p.csaleinvoiceid1,
        p.vdef__2 AS jinshuikaipiaoriqi,
        p.vdef__3 AS shijichukuriqi,
        p.name_2 AS kaipiaokehu,
        COALESCE(dep.name, p.name_5) AS xiaoshoubumen,
        CASE
            WHEN COALESCE(p.area_name2, '') IN ('四川省', '重庆市') THEN '川渝'
            ELSE '省外'
        END AS xiaoshoubumenduiyingdiqu,
        p.name_6 AS xiaoshouyewuyuan,
        p.sales_channel AS xiaoshouqudao,
        p.saleschannelsalesman AS qudaoyewuyuan,
        p.area_class AS diqufenlei,
        p.area_name2 AS shengshiqu,
        p.vdef__5 AS yetai,
        '普药非新特药' AS pinzhongleixing,
        p.material_name AS pinzhong,
        p.num AS pingzhenghao,
        p.material_code AS wuliaobianma,
        p.material_name AS wuliaomingcheng,
        p.material_spec AS guige,
        CAST(p.nnum AS NUMERIC(24,8)) AS zhushuliang,
        COALESCE(CAST(p.nnum AS NUMERIC(24,8)), 0) /
        NULLIF(CAST(NULLIF(SPLIT_PART(COALESCE(tb.measrate, ''), '/', 1), '') AS NUMERIC(24,8)), 0) AS jianshu,
        CAST(p.norigtaxprice1 AS NUMERIC(24,8)) AS zhuhanshuidanjia,
        CAST(p.norigtaxnetprice AS NUMERIC(24,8)) AS zhuhanshuijingjia,
        CAST(p.norigmny1 AS NUMERIC(24,8)) AS wushuijineheji,
        CAST(p.ntotalincomemny AS NUMERIC(24,8)) AS jiashuiheji,
        CAST(p.norigsubmny1 AS NUMERIC(24,8)) AS feiyongchongdijine,
        COALESCE(CAST(p.ntotalincomemny AS NUMERIC(24,8)), 0) + COALESCE(CAST(p.norigsubmny1 AS NUMERIC(24,8)), 0) AS chongdiqianjine,
        CAST(p.norigdiscount1 AS NUMERIC(24,8)) AS zhekoue,
        (
            COALESCE(CAST(p.ntotalincomemny AS NUMERIC(24,8)), 0) +
            COALESCE(CAST(p.norigsubmny1 AS NUMERIC(24,8)), 0)
        ) / 1.13 AS cdq_je_bhs,
        COALESCE(ld.price, 0) AS danweidijia,
        COALESCE(CAST(p.nnum AS NUMERIC(24,8)), 0) * COALESCE(ld.price, 0) AS dijiajine,
        p.cinvoicecustid,
        p.firstvbillcode,
        p.cmaterialvid,
        CURRENT_TIMESTAMP AS create_time,
        0 AS istatus
    FROM mis_auto_ptpcr p
    INNER JOIN temp_ns_sale_ids s
        ON s.id = p.id
    LEFT JOIN temp_mis_auto_tbdua tb
        ON tb.invoice_id = p.id
    LEFT JOIN mis_auto_yuorb dep
        ON dep.pk_vid = p.cdeptvid
    LEFT JOIN temp_mis_auto_ldttj ld
        ON ld.invoice_id = p.id;

    UPDATE public.mis_auto_jyycmx j
    SET
        fapiaoqijian = COALESCE(p.cperiod, v_cperiod),
        csaleinvoiceid1 = COALESCE(j.csaleinvoiceid1, p.csaleinvoiceid1),
        cinvoicecustid = p.cinvoicecustid,
        firstvbillcode = p.firstvbillcode,
        xiaoshoubumenduiyingdiqu = CASE
            WHEN COALESCE(j.shengshiqu, '') IN ('四川省', '四川', '重庆市', '重庆') THEN '川渝'
            ELSE '省外'
        END,
        pinzhongleixing = '新特药',
        pinzhong = COALESCE(j.pinzhong, j.wuliaomingcheng),
        cmaterialvid = p.cmaterialvid,
        cdq_je_bhs = CAST(COALESCE(j.chongdiqianjine, 0) AS NUMERIC(24,8)) / 1.13,
        create_time = CURRENT_TIMESTAMP,
        istatus = 0
    FROM temp_xt_sale_ids s
    LEFT JOIN mis_auto_ptpcr p
        ON p.id = s.id
    WHERE j.hesuanqijian = v_cperiod
      AND j.srcid ~ '^[0-9]+$'
      AND s.id = CAST(j.srcid AS BIGINT)
      AND COALESCE(j.pinzhongleixing, '') = '新特药';

    DROP TABLE IF EXISTS temp_ns_order_discount;
    CREATE TEMP TABLE temp_ns_order_discount AS
    SELECT
        p.id AS srcid,
        SUM(COALESCE(b.nnum, 0)) AS sum_nnum,
        SUM(COALESCE(b.ndiscount, 0)) AS sum_ndiscount
    FROM mis_auto_ptpcr p
    INNER JOIN temp_shared_sale_ids s
        ON s.id = p.id
    INNER JOIN mis_auto_emwtt o
        ON o.vbillcode = p.firstvbillcode
    INNER JOIN mis_auto_emwtt_sub_csaleorderbid b
        ON o.csaleorderid = b.csaleorderid
    WHERE p.cmaterialvid = b.cmaterialid
    GROUP BY p.id;

    DROP TABLE IF EXISTS temp_ns_invoice_material;
    CREATE TEMP TABLE temp_ns_invoice_material AS
    SELECT
        t.csaleinvoiceid1 AS csaleinvoiceid,
        t.wuliaobianma AS material_code,
        SUM(COALESCE(o.sum_ndiscount, 0)) AS sum_jine,
        SUM(COALESCE(o.sum_nnum, 0)) AS sum_shuliang
    FROM public.mis_auto_jyycmx t
    LEFT JOIN temp_ns_order_discount o
        ON CAST(t.srcid AS BIGINT) = o.srcid
    WHERE t.hesuanqijian = v_cperiod
      AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药')
    GROUP BY t.csaleinvoiceid1, t.wuliaobianma;

    WITH lpz AS (
        SELECT
            t.id,
            CASE
                WHEN COALESCE(im.sum_shuliang, 0) <> 0 THEN COALESCE(im.sum_jine, 0) / im.sum_shuliang
                ELSE 0
            END AS alloc_unit_discount,
            CASE
                WHEN COALESCE(t.danweidijia, 0) = 0 THEN 0
                ELSE (COALESCE(t.jiashuiheji, 0) - COALESCE(t.dijiajine, 0)) / 1.13
            END AS sw_lpz_sr_val,
            CASE
                WHEN COALESCE(t.danweidijia, 0) = 0 THEN 0
                ELSE (COALESCE(t.jiashuiheji, 0) - COALESCE(t.dijiajine, 0)) * 0.87115
            END AS zs_lpz_fy_val
        FROM public.mis_auto_jyycmx t
        LEFT JOIN temp_ns_invoice_material im
            ON t.csaleinvoiceid1 = im.csaleinvoiceid
           AND t.wuliaobianma = im.material_code
        WHERE t.hesuanqijian = v_cperiod
          AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药')
    )
    UPDATE public.mis_auto_jyycmx t
    SET dq_pz = ROUND(COALESCE(t.zhushuliang, 0) * COALESCE(lpz.alloc_unit_discount, 0) * -1, 8),
        dq_pz_bhs = ROUND((COALESCE(t.zhushuliang, 0) * COALESCE(lpz.alloc_unit_discount, 0) * -1) / 1.13, 8),
        sw_lpz_sr = ROUND(lpz.sw_lpz_sr_val, 8),
        lpz_sr_hj = ROUND(lpz.sw_lpz_sr_val, 8),
        sw_lpz_sr_hj = ROUND(lpz.sw_lpz_sr_val, 8),
        js_lpz_fy = ROUND(lpz.zs_lpz_fy_val, 8),
        zs_lpz_fy = ROUND(lpz.zs_lpz_fy_val, 8),
        bhdq_pz_cdq_bhs = ROUND(
            COALESCE(t.cdq_je_bhs, 0) +
            ((COALESCE(t.zhushuliang, 0) * COALESCE(lpz.alloc_unit_discount, 0) * -1) / 1.13) -
            COALESCE(lpz.zs_lpz_fy_val, 0),
            8
        )
    FROM lpz
    WHERE t.id = lpz.id;

    UPDATE public.mis_auto_jyycmx
    SET lpz_sr_hj = 0
    WHERE hesuanqijian = v_cperiod
      AND COALESCE(pinzhongleixing, '') IN ('普药非新特药', '新特药')
      AND COALESCE(lpz_sr_hj, 0) + COALESCE(chongdiqianjine, 0) = 0;

    IF v_cperiod_year <> '2025' THEN
        IF NOT EXISTS (
            SELECT 1
            FROM pg_proc p
            INNER JOIN pg_namespace n
                ON n.oid = p.pronamespace
            WHERE n.nspname = 'public'
              AND lower(p.proname) = lower('Cost_Data_Cost_Sales_Nospecialty_Drugs_JYYC_levels3')
              AND p.prokind = 'p'
        ) THEN
            result := -1;
            msg := 'Missing procedure public.Cost_Data_Cost_Sales_Nospecialty_Drugs_JYYC_levels3. Execute Cost_Data_Cost_Sales_Nospecialty_Drugs_JYYC_levels3.txt first.';
            RETURN;
        END IF;

        CALL public.Cost_Data_Cost_Sales_Nospecialty_Drugs_JYYC_levels3(v_cperiod, v_sub_result, v_sub_msg);
        IF COALESCE(v_sub_result, -1) <> 1 THEN
            result := -1;
            msg := CONCAT('Cross-period discount calculation failed: ', COALESCE(v_sub_msg, ''));
            RETURN;
        END IF;
        RAISE NOTICE 'levels3 finished';

        WITH tmp_fdq_agg AS (
            SELECT
                target_srcid,
                denominator,
                SUM(alloc_fdq_amount) * -1 AS alloc_fdq_amount,
                SUM(alloc_fdqb_amount) * -1 AS alloc_fdqb_amount
            FROM tmp_conversion_allocation_normal
            GROUP BY target_srcid, denominator
        )
        UPDATE public.mis_auto_jyycmx t
        SET fdq_ftamount = '0',
            fdq_ftnum = CAST(COALESCE(a.denominator, 0) AS VARCHAR),
            fdq_pz = ROUND(COALESCE(a.alloc_fdq_amount, 0), 8),
            fdq_pz_bcdq = ROUND(COALESCE(a.alloc_fdqb_amount, 0), 8)
        FROM tmp_fdq_agg a
        WHERE CAST(t.srcid AS BIGINT) = a.target_srcid
          AND t.hesuanqijian = v_cperiod
          AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药');

        WITH tmp_fdqb_agg AS (
            SELECT
                target_srcid,
                target_amount,
                SUM(alloc_fdq_amount) * -1 AS alloc_fdq_amount,
                SUM(alloc_fdqb_amount) * -1 AS alloc_fdqb_amount
            FROM tmp_conversion_allocation_fallback
            GROUP BY target_srcid, target_amount
        )
        UPDATE public.mis_auto_jyycmx t
        SET fdqb_ftamount = CAST(COALESCE(a.target_amount, 0) AS VARCHAR),
            fdqb_ftnum = '0',
            fdq_pz_jt = ROUND(COALESCE(a.alloc_fdq_amount, 0), 8),
            fdq_pz_bcdq_jt = ROUND(COALESCE(a.alloc_fdqb_amount, 0), 8)
        FROM tmp_fdqb_agg a
        WHERE CAST(t.srcid AS BIGINT) = a.target_srcid
          AND t.hesuanqijian = v_cperiod
          AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药');

        UPDATE public.mis_auto_jyycmx
        SET fdq_pz_hj = ROUND(COALESCE(fdq_pz, 0) + COALESCE(fdq_pz_jt, 0), 8),
            fdq_pz_bcdq_hj = ROUND(COALESCE(fdq_pz_bcdq, 0) + COALESCE(fdq_pz_bcdq_jt, 0), 8),
            fdq_pz_bcdq_bhs = ROUND((COALESCE(fdq_pz_bcdq, 0) + COALESCE(fdq_pz_bcdq_jt, 0)) / 1.13, 8),
            fdq_pz_bhs = ROUND((COALESCE(fdq_pz, 0) + COALESCE(fdq_pz_jt, 0)) / 1.13, 8)
        WHERE hesuanqijian = v_cperiod
          AND COALESCE(pinzhongleixing, '') IN ('普药非新特药', '新特药');
    END IF;

    UPDATE public.mis_auto_jyycmx
    SET pz_hj = ROUND(COALESCE(dq_pz, 0) + COALESCE(fdq_pz_bcdq_hj, 0) + COALESCE(fdq_pz_hj, 0), 8)
    WHERE hesuanqijian = v_cperiod
      AND COALESCE(pinzhongleixing, '') IN ('普药非新特药', '新特药');

    DROP TABLE IF EXISTS tmp_material_sales;
    CREATE TEMP TABLE tmp_material_sales AS
    SELECT
        p.cmaterialvid,
        SUM(COALESCE(CAST(p.nnum AS NUMERIC(24,8)), 0)) AS total_nnum
    FROM mis_auto_ptpcr p
    INNER JOIN temp_shared_sale_ids s
        ON s.id = p.id
    GROUP BY p.cmaterialvid;

    DROP TABLE IF EXISTS tmp_pz_sale1;
    CREATE TEMP TABLE tmp_pz_sale1 AS
    SELECT
        t.cmaterialvid,
        SUM(COALESCE(t.fdq_pz_hj, 0) + COALESCE(t.fdq_pz_bcdq_hj, 0)) AS total_pz
    FROM public.mis_auto_jyycmx t
    INNER JOIN mis_auto_ptpcr p
        ON p.id = CAST(t.srcid AS BIGINT)
       AND p.cmaterialvid = t.cmaterialvid
    LEFT JOIN mis_auto_yuorb y
        ON y.pk_vid = p.cdeptvid
    WHERE t.hesuanqijian = v_cperiod
      AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药')
      AND COALESCE(y.name, '') = '医疗事业部'
    GROUP BY t.cmaterialvid;

    DROP TABLE IF EXISTS tmp_saleamount;
    CREATE TEMP TABLE tmp_saleamount AS
    SELECT
        cperiod,
        material_code,
        saledeptcode,
        saledeptname,
        saleamount,
        hbtype
    FROM mis_auto_djmpl
    WHERE cperiod = v_cperiod;

    DROP TABLE IF EXISTS tmp_multi_group;
    CREATE TEMP TABLE tmp_multi_group AS
    SELECT
        saledeptcode,
        saleamount AS group_saleamount
    FROM tmp_saleamount
    WHERE COALESCE(hbtype, '') = '多部门';

    DROP TABLE IF EXISTS tmp_pz_multi;
    CREATE TEMP TABLE tmp_pz_multi AS
    SELECT
        mg.group_saleamount,
        SUM(COALESCE(t.fdq_pz_hj, 0) + COALESCE(t.fdq_pz_bcdq_hj, 0)) AS total_pz
    FROM public.mis_auto_jyycmx t
    INNER JOIN mis_auto_ptpcr p
        ON p.id = CAST(t.srcid AS BIGINT)
       AND p.cmaterialvid = t.cmaterialvid
    INNER JOIN mis_auto_yuorb y
        ON y.pk_vid = p.cdeptvid
    INNER JOIN tmp_multi_group mg
        ON mg.saledeptcode = y.code
    WHERE t.hesuanqijian = v_cperiod
      AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药')
    GROUP BY mg.group_saleamount;

    DROP TABLE IF EXISTS tmp_pz_other;
    CREATE TEMP TABLE tmp_pz_other AS
    SELECT
        p.cdeptvid,
        SUM(COALESCE(t.fdq_pz_hj, 0) + COALESCE(t.fdq_pz_bcdq_hj, 0)) AS total_pz
    FROM public.mis_auto_jyycmx t
    LEFT JOIN mis_auto_ptpcr p
        ON p.id = CAST(t.srcid AS BIGINT)
       AND p.cmaterialvid = t.cmaterialvid
    LEFT JOIN mis_auto_yuorb y
        ON y.pk_vid = p.cdeptvid
    WHERE t.hesuanqijian = v_cperiod
      AND COALESCE(t.pinzhongleixing, '') IN ('普药非新特药', '新特药')
      AND y.code IN (
          SELECT saledeptcode
          FROM mis_auto_djmpl
          WHERE cperiod = v_cperiod
            AND COALESCE(hbtype, '') = '单部门'
      )
    GROUP BY p.cdeptvid;

    DROP TABLE IF EXISTS tmp_dept_hytype;
    CREATE TEMP TABLE tmp_dept_hytype AS
    SELECT DISTINCT ON (saledeptcode)
        saledeptcode,
        hbtype
    FROM mis_auto_djmpl
    WHERE cperiod = v_cperiod
      AND saledeptcode IS NOT NULL
    ORDER BY saledeptcode;
